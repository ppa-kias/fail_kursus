// Kod Apps Script Google Sheets & Drive Backend KIAS
function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === "loadAll") {
    var sessionSheet = ss.getSheetByName("SESI_AKADEMIK") || createSessionSheet(ss);
    var sessionRows = sessionSheet.getDataRange().getValues();
    var sessions = [];
    for (var i = 1; i < sessionRows.length; i++) {
      if (sessionRows[i][0]) sessions.push(sessionRows[i][0].toString().trim());
    }
    
    var jsonSheet = ss.getSheetByName("DATABASE_JSON") || ss.insertSheet("DATABASE_JSON");
    var jsonRows = jsonSheet.getDataRange().getValues();
    var payloadMap = {};
    for (var j = 1; j < jsonRows.length; j++) {
      if (jsonRows[j][0]) {
        payloadMap[jsonRows[j][0]] = JSON.parse(jsonRows[j][1]);
      }
    }
    
    // Tarik draf akaun berdaftar
    var accSheet = ss.getSheetByName("USERS_ACCOUNTS") || ss.insertSheet("USERS_ACCOUNTS");
    var accRows = accSheet.getDataRange().getValues();
    var accounts = [];
    if (accRows.length > 1) {
      for (var k = 1; k < accRows.length; k++) {
        if (accRows[k][0]) {
          accounts.push({
            username: accRows[k][0],
            password: accRows[k][1],
            name: accRows[k][2],
            fakulti: accRows[k][3],
            jawatan: accRows[k][4]
          });
        }
      }
    }
    
    var responseData = {
      academicSessions: sessions,
      payloadMap: payloadMap,
      accounts: accounts
    };
    
    return ContentService.createTextOutput(JSON.stringify({status: "success", data: responseData}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Action tidak sah."}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var postData = JSON.parse(e.postData.contents);
    var action = postData.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (action === "saveAll") {
      // Simpan Sesi Akademik
      var sessionSheet = ss.getSheetByName("SESI_AKADEMIK") || ss.insertSheet("SESI_AKADEMIK");
      sessionSheet.clear();
      sessionSheet.appendRow(["SESI_NAMA"]);
      postData.academicSessions.forEach(function(s) {
        sessionSheet.appendRow([s]);
      });
      
      // Simpan Peta Data JSON
      var jsonSheet = ss.getSheetByName("DATABASE_JSON") || ss.insertSheet("DATABASE_JSON");
      jsonSheet.clear();
      jsonSheet.appendRow(["SESI_KEY", "JSON_DATA"]);
      Object.keys(postData.payloadMap).forEach(function(key) {
        jsonSheet.appendRow([key, JSON.stringify(postData.payloadMap[key])]);
      });
      
      // Simpan Data Akaun
      if (postData.accounts) {
        var accSheet = ss.getSheetByName("USERS_ACCOUNTS") || ss.insertSheet("USERS_ACCOUNTS");
        accSheet.clear();
        accSheet.appendRow(["USERNAME", "PASSWORD", "NAME", "FAKULTI", "JAWATAN"]);
        postData.accounts.forEach(function(acc) {
          accSheet.appendRow([acc.username, acc.password, acc.name, acc.fakulti, acc.jawatan]);
        });
      }
      
      return ContentService.createTextOutput(JSON.stringify({status: "success"}))
        .setMimeType(ContentService.MimeType.JSON);
    } 
    
    // Tindakan Baru: Memuat Naik Fail Terus Ke Google Drive
    else if (action === "uploadFile") {
      var folderName = "SISTEM_FAIL_KURSUS_LAMPIRAN";
      var folders = DriveApp.getFoldersByName(folderName);
      var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
      
      var fileBytes = Utilities.base64Decode(postData.base64Data);
      var blob = Utilities.newBlob(fileBytes, postData.mimeType, postData.filename);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        fileUrl: file.getUrl(),
        fileName: postData.filename
      })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Kod Apps Script Google Sheets Backend KIAS (Lengkap doGet dan doPost)
function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === "loadAll") {
    var sessionSheet = ss.getSheetByName("SESI_AKADEMIK") || createSessionSheet(ss);
    var sessionRows = sessionSheet.getDataRange().getValues();
    var sessions = [];
    for (var i = 1; i < sessionRows.length; i++) {
      if (sessionRows[i][0]) sessions.push(sessionRows[i][0].toString().trim());
    }
    
    var jsonSheet = ss.getSheetByName("DATABASE_JSON") || ss.insertSheet("DATABASE_JSON");
    var jsonRows = jsonSheet.getDataRange().getValues();
    var payloadMap = {};
    for (var j = 1; j < jsonRows.length; j++) {
      if (jsonRows[j][0]) {
        payloadMap[jsonRows[j][0]] = JSON.parse(jsonRows[j][1]);
      }
    }
    
    var responseData = {
      academicSessions: sessions,
      payloadMap: payloadMap
    };
    
    return ContentService.createTextOutput(JSON.stringify({status: "success", data: responseData}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Action tidak sah."}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var postData = JSON.parse(e.postData.contents);
    var action = postData.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (action === "saveAll") {
      var sessionSheet = ss.getSheetByName("SESI_AKADEMIK") || createSessionSheet(ss);
      sessionSheet.clear();
      sessionSheet.appendRow(["Sesi Akademik"]);
      var sessions = postData.academicSessions || [];
      sessions.forEach(function(s) {
        if (s) sessionSheet.appendRow([s]);
      });
      
      var jsonSheet = ss.getSheetByName("DATABASE_JSON") || ss.insertSheet("DATABASE_JSON");
      jsonSheet.clear();
      jsonSheet.appendRow(["Sesi_Akademik", "Payload_JSON"]);
      
      var payloadMap = postData.payloadMap || {};
      Object.keys(payloadMap).forEach(function(sessionKey) {
        jsonSheet.appendRow([sessionKey, JSON.stringify(payloadMap[sessionKey])]);
      });
      
      return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Data berjaya disimpan!"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Action tidak disokong."}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function createSessionSheet(ss) {
  var sheet = ss.insertSheet("SESI_AKADEMIK");
  sheet.appendRow(["Sesi Akademik"]);
  sheet.appendRow(["2025/2026 - SEM 2"]);
  sheet.appendRow(["2025/2026 - SEM 1"]);
  sheet.appendRow(["2024/2025 - SEM 2"]);
  sheet.appendRow(["2024/2025 - SEM 1"]);
  return sheet;
}

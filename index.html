Cara Setup Google Sheet Anda Sebagai Pangkalan Data

Format Google Sheets ini menyokong penyimpanan berbilang subjek bagi seorang pensyarah secara fleksibel. Tampalkan kod Google Apps Script yang disediakan ke dalam editor Google Sheet anda.

GoogleAppsScript.js 
// Kod Apps Script Google Sheets Backend KIAS - (Dual Mode: JSON + Paparan Manusia)
function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === "loadAll") {
    // 1. Ambil draf sesi
    var sessionSheet = ss.getSheetByName("SESI_AKADEMIK") || createSessionSheet(ss);
    var sessionRows = sessionSheet.getDataRange().getValues();
    var sessions = [];
    for (var i = 1; i < sessionRows.length; i++) {
      if (sessionRows[i][0]) sessions.push(sessionRows[i][0].toString().trim());
    }
    
    // 2. Ambil data JSON pangkalan data
    var jsonSheet = ss.getSheetByName("DATABASE_JSON") || ss.insertSheet("DATABASE_JSON");
    var jsonRows = jsonSheet.getDataRange().getValues();
    var payloadMap = {};
    for (var j = 1; j < jsonRows.length; j++) {
      if (jsonRows[j][0]) {
        payloadMap[jsonRows[j][0]] = JSON.parse(jsonRows[j][1]);
      }
    }
    
    var responseData = {
      academicSessions: sessions,
      payloadMap: payloadMap
    };
    
    return ContentService.createTextOutput(JSON.stringify({status: "success", data: responseData}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader('Access-Control-Allow-Origin', '*');
  }
  
  return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Action doGet tidak sah."}))
    .setMimeType(ContentService.MimeType.JSON)
    .setHeader('Access-Control-Allow-Origin', '*');
}

function doPost(e) {
  try {
    var params = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (params.action === "saveAll") {
      // 1. Simpan Sesi Akademik
      var sessionSheet = ss.getSheetByName("SESI_AKADEMIK") || createSessionSheet(ss);
      sessionSheet.clearContents();
      sessionSheet.appendRow(["Sesi Akademik / Semester"]);
      params.academicSessions.forEach(function(s) {
        sessionSheet.appendRow([s]);
      });
      
      // 2. Simpan Database JSON (Integriti Tinggi)
      var jsonSheet = ss.getSheetByName("DATABASE_JSON") || ss.insertSheet("DATABASE_JSON");
      jsonSheet.clearContents();
      jsonSheet.appendRow(["Sesi", "JSON_Data"]);
      
      Object.keys(params.payloadMap).forEach(function(session) {
        jsonSheet.appendRow([session, JSON.stringify(params.payloadMap[session])]);
      });
      
      // 3. Kemaskini Flat Table Paparan Manusia (Mudah Dibaca)
      updateHumanReadableSheet(ss, params.payloadMap);
      
      return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Database berjaya disegerakkan."}))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader('Access-Control-Allow-Origin', '*');
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader('Access-Control-Allow-Origin', '*');
  }
}

function createSessionSheet(ss) {
  var s = ss.insertSheet("SESI_AKADEMIK");
  s.appendRow(["Sesi Akademik / Semester"]);
  s.appendRow(["2025/2026 - SEM 2"]);
  s.appendRow(["2025/2026 - SEM 1"]);
  return s;
}

// Menjana flat table automatik mengikut standard 10 folder bagi setiap subjek pensyarah
function updateHumanReadableSheet(ss, payloadMap) {
  var sheet = ss.getSheetByName("PAPARAN_KESELURUHAN") || ss.insertSheet("PAPARAN_KESELURUHAN");
  sheet.clearContents();
  
  // Header lajur flat-table
  var headers = [
    "Sesi Akademik", "Nama Pensyarah", "Jawatan / Status", "Fakulti / Jabatan", "Kod Kursus", "Progres",
    "F1", "F2.1", "F2.2", "F2.3", "F3.1", "F3.2", "F3.3", "F4", "F5.1", "F5.2", "F6", "F7.1", "F7.2", "F8", "F9", "F10",
    "Catatan / Ulasan Khas"
  ];
  sheet.appendRow(headers);
  
  Object.keys(payloadMap).forEach(function(session) {
    var data = payloadMap[session];
    if (data && data.lecturers) {
      data.lecturers.forEach(function(l) {
        if (l.senaraiKursus) {
          l.senaraiKursus.forEach(function(k) {
            // Pengiraan progres
            var f = k.folder || {};
            var countYA = 0, countTIDAK = 0;
            var fKeys = ["f1", "f2_1", "f2_2", "f2_3", "f3_1", "f3_2", "f3_3", "f4", "f5_1", "f5_2", "f6", "f7_1", "f7_2", "f8", "f9", "f10"];
            
            fKeys.forEach(function(key) {
              if (f[key] === 'YA') countYA++;
              if (f[key] === 'TIDAK') countTIDAK++;
            });
            var progressPercent = (countYA + countTIDAK === 0) ? 100 : Math.round((countYA / (countYA + countTIDAK)) * 100);
            
            sheet.appendRow([
              session, l.nama, l.jawatan, l.fakulti, k.kodKursus, progressPercent + "%",
              f.f1 || 'TIDAK', f.f2_1 || 'TIDAK', f.f2_2 || 'TIDAK', f.f2_3 || 'TIDAK',
              f.f3_1 || 'TIDAK', f.f3_2 || 'TIDAK', f.f3_3 || 'TIDAK', f.f4 || 'TIDAK',
              f.f5_1 || 'TIDAK', f.f5_2 || 'TIDAK', f.f6 || 'TIDAK', f.f7_1 || 'TIDAK',
              f.f7_2 || 'TIDAK', f.f8 || 'TIDAK', f.f9 || 'TIDAK', f.f10 || 'TIDAK',
              k.catatan || ""
            ]);
          });
        }
      });
    }
  });
  
  // Format visual header jadual agar kelihatan premium
  sheet.getRange(1, 1, 1, headers.length).setBackground("#065f46").setFontColor("#ffffff").setFontWeight("bold");
  sheet.autoResizeColumns(1, headers.length);
}
